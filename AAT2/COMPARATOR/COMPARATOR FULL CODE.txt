module cmp_4bit (
    input wire [3:0] A,
    input wire [3:0] B,
    output reg A_gt_B,
    output reg A_lt_B,
    output reg A_et_B
);

always @(*) begin
    A_gt_B = 1'b0;
    A_lt_B = 1'b0;
    A_et_B = 1'b0;

    if (A > B)
        A_gt_B = 1'b1;
    else if (A < B)
        A_lt_B = 1'b1;
    else
        A_et_B = 1'b1;
end

endmodule

module tb_4bit_cmp;

reg [3:0] A;
reg [3:0] B;

wire A_gt_B;
wire A_lt_B;
wire A_et_B;

cmp_4bit uut (
    .A(A),
    .B(B),
    .A_gt_B(A_gt_B),
    .A_lt_B(A_lt_B),
    .A_et_B(A_et_B)
);

initial begin
    A = 4'd5;
    B = 4'd5;
    #10;

    #10;

    A = 4'd2;
    B = 4'd9;
    #10;

    A = 4'd15;
    B = 4'd15;
    #10;

    $finish;
end

endmodule
