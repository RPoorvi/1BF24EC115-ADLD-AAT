module usr_4bit (
    input wire clk,
    input wire rst,
    input wire [1:0] s,
    input wire [3:0] d,
    input wire msb_in,
    input wire lsb_in,
    output reg [3:0] q
);

always @(posedge clk or posedge rst) begin
    if (rst)
        q <= 4'b0000;
    else begin
        case (s)
            2'b00: q <= q;
            2'b01: q <= {msb_in, q[3:1]};
            2'b10: q <= {q[2:0], lsb_in};
            2'b11: q <= d;
            default: q <= q;
        endcase
    end
end

endmodule

module tb_usr_4bit;

reg clk, rst;
reg [1:0] s;
reg [3:0] d;
reg msb_in, lsb_in;
wire [3:0] q;

usr_4bit uut (
    .clk(clk),
    .rst(rst),
    .s(s),
    .d(d),
    .msb_in(msb_in),
    .lsb_in(lsb_in),
    .q(q)
);

always #5 clk = ~clk;

initial begin
    clk = 0;
    rst = 1;
    s = 2'b00;
    d = 4'b0000;
    msb_in = 0;
    lsb_in = 0;

    #10 rst = 0;

    s = 2'b11;
    d = 4'b1011;
    #10;

    s = 2'b00;
    #10;

    s = 2'b01;
    msb_in = 1;
    #10;

    msb_in = 0;
    #10;

    s = 2'b10;
    lsb_in = 1;
    #10;

    lsb_in = 0;
    #10;

    #20 $finish;
end

endmodule
